title: Join and leave announcer
description: "A live region that tells screen reader users when a peer joins or leaves."

---

A screen reader user cannot see the bar change. `CursorLayer` cannot help: each cursor root has `aria-hidden="true"`, so screen readers skip it. Add a live region that says when a peer joins or leaves.

`usePeerChanges` calls your listener for each join, leave, and `user` or `context` update. If ten peers join in one second, ten messages would queue up in the screen reader. The component below collects the changes for 1 second and announces one message:

```tsx
import { usePeerChanges, type PeerChange } from "@anchor-presence/react";
import { useEffect, useRef, useState } from "react";

type Person = { name: string };

const BURST_MS = 1000;
const CLEAR_MS = 100;

const people = (count: number) => `${count} ${count === 1 ? "person" : "people"}`;

function describeChanges(changes: PeerChange<unknown, Person>[]): string {
  const [only] = changes;
  if (changes.length === 1 && only !== undefined)
    return `${only.user?.name ?? "Someone"} ${only.type === "join" ? "joined" : "left"}`;
  const joined = changes.filter(({ type }) => type === "join").length;
  const left = changes.length - joined;
  return [joined > 0 && `${people(joined)} joined`, left > 0 && `${people(left)} left`]
    .filter(Boolean)
    .join(", ");
}

function PeerAnnouncer() {
  const [message, setMessage] = useState("");
  const pending = useRef<PeerChange<unknown, Person>[]>([]);
  const timer = useRef<ReturnType<typeof setTimeout>>(undefined);
  const announceTimer = useRef<ReturnType<typeof setTimeout>>(undefined);

  usePeerChanges<unknown, Person>((change) => {
    if (change.type === "update") return;
    pending.current.push(change);
    timer.current ??= setTimeout(() => {
      const text = describeChanges(pending.current);
      pending.current = [];
      timer.current = undefined;
      // A region does not announce a message equal to the last one, so clear it first.
      setMessage("");
      announceTimer.current = setTimeout(() => setMessage(text), CLEAR_MS);
    }, BURST_MS);
  });

  useEffect(
    () => () => {
      clearTimeout(timer.current);
      clearTimeout(announceTimer.current);
    },
    [],
  );

  return (
    <div role="status" aria-live="polite" className="sr-only">
      {message}
    </div>
  );
}
```

Render `PeerAnnouncer` once, inside `PresenceProvider`. Keep it mounted: a screen reader only announces text that changes inside a region that already exists. `sr-only` is the Tailwind class that hides the text from the eye and keeps it for screen readers. Use an equal CSS rule if you do not use Tailwind.

- The first change starts a 1 second window. Every later join or leave in that window joins the same message.
- The component clears the region for 100 ms before it sets the text. Then two windows with the same text, like two "Ada joined" messages, are both announced.
- A window with two or more changes says "3 people joined, 1 person left".
- The listener ignores `update`, so a change of `user` or `context` stays silent.
- `user.name` is untrusted network data. React renders it as text.
